% ─── Titres de sections ─────────────────────────────────────────────────────
\titleformat{\section}
  {\normalfont\large\bfseries\sffamily\color{viteText}}
  {}
  {0em}
  {\textcolor{vitePurple}{\rule[-3pt]{4pt}{1.5em}}\hspace{8pt}}
  [\vspace{1pt}\textcolor{vitePurple}{\titlerule[1pt]}]

  {\normalfont\normalsize\bfseries\sffamily\color{viteText}}
  {}
  {0em}

\titlespacing*{\section}{0pt}{20pt}{10pt}

% ─── Boîtes ─────────────────────────────────────────────────────────────────
\tcbset{
  vitebase/.style={
    enhanced, breakable, sharp corners,
    left=10pt, right=10pt, top=7pt, bottom=7pt,
    boxrule=0pt, frame hidden,
    fontupper=\small\sffamily\color{viteText},
  },
  infobox/.style={
    vitebase,
    borderline west={3pt}{0pt}{vitePurple},
    colback=vitePurpleGlow,
  },
  warnbox/.style={
    vitebase,
    borderline west={3pt}{0pt}{viteRed},
    colback=viteRedBg,
  },
  successbox/.style={
    vitebase,
    borderline west={3pt}{0pt}{viteGreen},
    colback=viteGreenBg,
  },
  notebox/.style={
    vitebase,
    borderline west={3pt}{0pt}{viteAmber},
    colback=viteAmberBg,
  },
}

\newenvironment{infobox}{\begin{tcolorbox}[infobox]}{\end{tcolorbox}}
\newenvironment{warnbox}{\begin{tcolorbox}[warnbox]}{\end{tcolorbox}}
\newenvironment{successbox}{\begin{tcolorbox}[successbox]}{\end{tcolorbox}}
\newenvironment{notebox}{\begin{tcolorbox}[notebox]}{\end{tcolorbox}}
% Same signature as teal's ticketbox, so templates work with both preambles
\newtcolorbox{ticketbox}[2][]{
  vitebase, borderline west={3pt}{0pt}{viteCyan}, colback=viteSurface,
  title={#2}, coltitle=viteCyan, colbacktitle=viteSurface,
  fonttitle=\bfseries\sffamily, #1
}

% ─── Page de titre ───────────────────────────────────────────────────────────
% Usage : \makeViteDarkTitle{Titre principal}{Sous-titre}{Contenu encart périmètre}
\newcommand{\makeViteDarkTitle}[3]{%
  \begin{tcolorbox}[
    enhanced, sharp corners,
    colback=viteSurface, colframe=viteBorder,
    boxrule=0.6pt,
    left=18pt, right=18pt, top=22pt, bottom=18pt,
    borderline north={3pt}{0pt}{vitePurple},
    borderline south={1pt}{0pt}{viteBorder},
  ]
    \sffamily\color{viteText}
    {\footnotesize\textcolor{viteTextMuted}{\MakeUppercase{\reportorg}}}
    \vspace{10pt}

    {\fontsize{24}{28}\selectfont\bfseries
      \textcolor{vitePurple}{#1}}

    \vspace{4pt}
    {\large\textcolor{viteTextMuted}{#2}}

    \vspace{14pt}
    \textcolor{viteBorder}{\rule{\linewidth}{0.4pt}}
    \vspace{8pt}

    {\small\color{viteText} #3}
  \end{tcolorbox}
  \vspace{20pt}
}
